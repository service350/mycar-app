var extraUnitWaiver = (e, cat, young) => pickPrice(e, "waiverPrice", "waiverByCategory", cat, young, "youngWaiverPrice", "waiverByCategoryYoung");
function packageExtras(p, cat) {
  return (p.extras || []).filter((e) => e && e.enabled !== false && (!(e.categories || []).length || cat && e.categories.includes(cat.id)));
}
function extraQty(e, hours, km) {
  if (e.unit === "hour") return hours;
  if (e.unit === "day") return Math.max(1, Math.ceil(hours / 24 - 1e-9));
  if (e.unit === "km") return km;
  return 1;
}
function packageQuote(p, cat, reqHours, dist, start, wantWaiver, young, chosenOpt) {
  if (!packageLive(p) || !cat) return null;
  if (young && !youngWaiverOk()) wantWaiver = false;
  const cats = packageCats(p);
  if (cats && !cats.includes(cat.id)) return null;
  const extras = packageExtras(p, cat);
  const before = extras.find((e) => e.mode === "before");
  const after = extras.find((e) => e.mode === "after");
  let kBefore = 0;
  if (!startAllowed(p, start)) {
    if (!before || !start) return null;
    const max = Number(before.max) || 0;
    for (let k = 1; k <= max; k++) {
      if (startAllowed(p, new Date(start.getTime() + k * 36e5))) {
        kBefore = k;
        break;
      }
    }
    if (!kBefore) return null;
  }
  const rest = reqHours - kBefore;
  if (rest <= 0) return null;
  let kAfter = 0;
  if (rest > packageHours(p)) {
    if (!after) return null;
    kAfter = Math.ceil(rest - packageHours(p) - 1e-9);
    if (kAfter > (Number(after.max) || 0)) return null;
  }
  const rate = hasVal(p.extraKmPrice) ? Number(p.extraKmPrice) : Number(cat.rates.daily.kmPrice) || 0;
  if (young && !packageYoungAllowed(p)) return null;
  const youngFee = young ? packageYoungFee(p, cat) : 0;
  let waiver = 0;
  if (wantWaiver) {
    const w = packageWaiverPrice(p, cat, young);
    if (w === null) return null;
    waiver = w;
  }
  let best = null;
  packageTracks(p).forEach((t) => {
    const over = t.unlimited ? 0 : Math.max(0, dist - (Number(t.km) || 0));
    const cost = trackPrice(t, cat) + over * rate;
    if (!best || cost < best.cost) best = { cost, track: t, over };
  });
  if (best) {
    const lines = [];
    const add = (e, qty) => {
      if (!qty) return;
      const unit = extraUnitPrice(e, cat, young) + (wantWaiver ? extraUnitWaiver(e, cat, young) : 0);
      lines.push({ e, qty, cost: unit * qty, withWaiver: wantWaiver && extraUnitWaiver(e, cat, young) > 0 });
    };
    if (before) add(before, kBefore);
    if (after) add(after, kAfter);
    extras.filter((e) => e.mode === "always").forEach((e) => add(e, extraQty(e, reqHours, dist)));
    extras.filter((e) => e.mode === "optional" && chosenOpt && chosenOpt[e.id]).forEach((e) => add(e, extraQty(e, reqHours, dist)));
    const extrasSum = lines.reduce((t, x) => t + x.cost, 0);
    best.base = best.cost;
    best.waiver = waiver;
    best.youngFee = youngFee;
    best.extras = lines;
    best.kBefore = kBefore;
    best.kAfter = kAfter;
    best.optional = extras.filter((e) => e.mode === "optional");
    best.cost += waiver + youngFee + extrasSum;
  }
  return best;
}

var DEFAULT_HAPPY_WINDOWS = [

function happyWindows() {
  const w = (cfg().features || {}).happyWindows;
  return Array.isArray(w) ? w : DEFAULT_HAPPY_WINDOWS;

function inWindow(w, d) {
  const wd = d.getDay(), mins = d.getHours() * 60 + d.getMinutes();
  if (w.type === "span") {
    const f2 = hm(w.fromTime), t2 = hm(w.toTime);
    if (f2 === null || t2 === null) return false;
    const a = Number(w.fromDay) * 1440 + f2, b = Number(w.toDay) * 1440 + t2, x = wd * 1440 + mins;
    return a <= b ? x >= a && x < b : x >= a || x < b;
  }
  const f = hm(w.from), t = hm(w.to);
  if (f === null || t === null) return false;
  const days = (w.days || []).map(Number);
  if (f < t) return days.includes(wd) && mins >= f && mins < t;
  return days.includes(wd) && mins >= f || days.includes((wd + 6) % 7) && mins < t;
}
function happyHourCount(start, hours) {
  const ws = happyWindows();
  let n = 0;
  for (let i = 0; i < hours; i++) {
    const t = new Date(start.getTime() + i * 36e5 + 18e5);
    if (ws.some((w) => inWindow(w, t))) n++;
  }
  return n;
}
function calculateCost(catId, prd, dur, dist, hasWaiver, driverIsUnder24, happyHourActive, happyCount) {
  const c = cfg();
  const baseCat = c.pricing.categories.find((x) => x.id === catId);
  if (!baseCat) return { total: 0, timeCost: 0, kmCost: 0, rawKmCost: 0, isCapped: false, tierLabel: "", waiverCost: 0, youthSurcharge: 0 };
  const activeRates = baseCat.rates[prd];
  const happyAll = baseCat.happyHourRate && prd === "hourly" && happyHourActive;
  let timeCost;
  if (happyAll && happyCount != null && happyCount < dur) {
    const hh = Math.max(0, happyCount);
    timeCost = (dur - hh) * activeRates.timePrice + hh * baseCat.happyHourRate;
  } else {
    timeCost = dur * (happyAll ? baseCat.happyHourRate : activeRates.timePrice);
  }
  let kmCost = 0, isCapped = false, rawKmCost = 0, tierLabel = "";
  if (prd === "hourly") {
    const tiers = activeRates.kmTiers;
    let idx = tiers.length - 1;
    for (let i = 0; i < tiers.length - 1; i++) {
      if (dist <= tiers[i].max) {
        idx = i;
        break;
      }
    }
    const tier = tiers[idx];
    const last = idx === tiers.length - 1;
    const from = idx === 0 ? 0 : tiers[idx - 1].max + 1;
    rawKmCost = dist * tier.rate;
    tierLabel = txp(last ? "res.tierLabelLast" : "res.tierLabel", { "\u05DE\u05E1\u05E4\u05E8": idx + 1, "\u05D8\u05D5\u05D5\u05D7": last ? `${from}+` : `${from}-${tier.max}`, "\u05EA\u05E2\u05E8\u05D9\u05E3": fmtN(tier.rate) });
    if (hasVal(tier.cap) && rawKmCost > tier.cap) {
      kmCost = Number(tier.cap);
      isCapped = true;
    } else {
      kmCost = rawKmCost;
    }
  } else {
    kmCost = dist * activeRates.kmPrice;
    rawKmCost = kmCost;
    tierLabel = txp("res.fixedKm", { "\u05EA\u05E2\u05E8\u05D9\u05E3": fmtN(activeRates.kmPrice) });
  }
  let waiverCost = 0;
  if (hasWaiver && baseCat.deductibleWaiver && baseCat.deductibleWaiver[prd]) waiverCost = dur * baseCat.deductibleWaiver[prd];
  let youthSurcharge = 0;
  if (driverIsUnder24) youthSurcharge = dur * (c.pricing.youthSurcharges[prd] || 0);
  return { total: timeCost + kmCost + waiverCost + youthSurcharge, timeCost, kmCost, rawKmCost, isCapped, tierLabel, waiverCost, youthSurcharge, period: prd };
}

var DEFAULT_URLS = "https://service350.github.io/mycar-app/mycar-config.js,https://service350.github.io/mycar-app/app/config.js";
var CFG = null;
var CFG_AT = 0;
async function loadConfig(env) {
  if (CFG && Date.now() - CFG_AT < 6e5) return CFG;
  const urls = String(env.CONFIG_URLS || DEFAULT_URLS).split(",").map((u) => u.trim()).filter(Boolean);
  let best = null;
  for (const u of urls) {
    try {
      const r = await fetch(u + (u.includes("?") ? "&" : "?") + "t=" + Date.now(), { cf: { cacheTtl: 0 } });
      if (!r.ok) continue;
      const s = await r.text();
      const i = s.indexOf("{"), j = s.lastIndexOf("}");
      const c = JSON.parse(s.slice(i, j + 1));
      if (c && c.pricing && (!best || Number(c.publishedAt || 0) > Number(best.publishedAt || 0))) best = c;
    } catch (e) {
    }
  }
  if (best) {
    CFG = best;
    CFG_AT = Date.now();
    setConfig2(best);
  }
  if (!CFG) throw new Error("no price list");
  return CFG;
}
async function verifySignature(req, body, secret) {
  if (!secret) return true;
  const sig = req.headers.get("x-hub-signature-256") || "";
  const key = await crypto.subtle.importKey("raw", new TextEncoder().encode(secret), { name: "HMAC", hash: "SHA-256" }, false, ["sign"]);
  const mac = await crypto.subtle.sign("HMAC", key, new TextEncoder().encode(body));
  const hex = [...new Uint8Array(mac)].map((b) => b.toString(16).padStart(2, "0")).join("");
  return sig === "sha256=" + hex;
}
function toWhatsApp(to, r) {
  let text = r.text || "";
  if (r.url) text += `

${r.url.title}: ${r.url.href}`;
  const base = { messaging_product: "whatsapp", recipient_type: "individual", to };
  const out = [];
  const interactive = (body, action, type) => out.push({ ...base, type: "interactive", interactive: { type, body: { text: body }, action } });
  if (r.buttons && r.buttons.length) {
    const action = { buttons: r.buttons.slice(0, 3).map((b) => ({ type: "reply", reply: { id: b.id, title: b.title.slice(0, 20) } })) };
    if (text.length <= 1024) interactive(text, action, "button");
    else {
      out.push({ ...base, type: "text", text: { body: text.slice(0, 4096), preview_url: true } });
      interactive("\u{1F447}", action, "button");
    }
  } else if (r.list && r.list.rows.length) {
    interactive(text.slice(0, 1024), { button: r.list.button.slice(0, 20), sections: [{ title: r.list.button.slice(0, 24), rows: r.list.rows.slice(0, 10).map((x) => ({ id: x.id, title: x.title.slice(0, 24), ...x.description ? { description: x.description.slice(0, 72) } : {} })) }] }, "list");
  } else {
    out.push({ ...base, type: "text", text: { body: text.slice(0, 4096), preview_url: true } });
  }
  return out;
}

async function onMessage(env, m) {
  let msg = null;
  if (m.type === "text") msg = { text: m.text.body };
  else if (m.type === "interactive") msg = { id: (m.interactive.button_reply || m.interactive.list_reply || {}).id };
  else if (m.type === "button") msg = { text: m.button.text };
  if (!msg || !msg.text && !msg.id) return;
  await loadConfig(env);
  const key = "u:" + m.from;
  const raw = env.STATE ? await env.STATE.get(key) : null;
  const state = raw ? JSON.parse(raw) : null;
  const { state: next, replies } = handle(state, msg);
  if (env.STATE) await env.STATE.put(key, JSON.stringify(next), { expirationTtl: 86400 });
  for (const r of replies) for (const p of toWhatsApp(m.from, r)) await send(env, p);
}
var worker_default = {
  async fetch(req, env, ctx) {
    const url = new URL(req.url);
    if (req.method === "GET") {
      if (url.searchParams.get("hub.mode") === "subscribe" && url.searchParams.get("hub.verify_token") === env.VERIFY_TOKEN) return new Response(url.searchParams.get("hub.challenge") || "", { status: 200 });
      if (url.pathname === "/health") {
        try {
          const c = await loadConfig(env);
          return Response.json({ ok: true, cars: c.pricing.categories.length, publishedAt: c.publishedAt || null });
        } catch (e) {
          return Response.json({ ok: false, error: e.message }, { status: 500 });
        }
      }
      return new Response("My Car bot", { status: 200 });
    }
    if (req.method !== "POST") return new Response("", { status: 405 });
    const body = await req.text();
    if (!await verifySignature(req, body, env.APP_SECRET)) return new Response("bad signature", { status: 401 });
    let data;
    try {
      data = JSON.parse(body);
    } catch (e) {
      return new Response("", { status: 400 });
    }
    const msgs = [];
    (data.entry || []).forEach((e) => (e.changes || []).forEach((c) => ((c.value || {}).messages || []).forEach((m) => msgs.push(m))));
    ctx.waitUntil(Promise.all(msgs.map((m) => onMessage(env, m).catch((err) => console.log("error", err && err.message)))));
    return new Response("ok", { status: 200 });
  }
};
